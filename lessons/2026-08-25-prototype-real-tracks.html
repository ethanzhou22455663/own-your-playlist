Concepts touched

  • Prototype integration: taking a standalone design mock and making it the real application UI.
  • IPC arguments: data can be passed from renderer → preload → main process through the arguments of ipcRenderer.invoke and ipcMain.handle.
  • Parameter naming: folderPath is just a chosen name; it must match across preload and main process.
  • _event: the first argument to an ipcMain.handle callback is the event object; prefixing with _ means "intentionally unused."
  • Default values: folderPath || './music' uses the provided path, or ./music if it is empty.
  • Unique ID: because scanned tracks have no built-in id, filePath is used as the unique identifier.
  • Row number vs metadata trackNumber: the # column shows the list position (index + 1) for a clean playlist view; the embedded trackNumber tag is kept for editing.
  • Array formatting: arr.join(', ') turns ['流行', '摇滚'] into "流行, 摇滚".
  • Duration formatting: Math.floor(seconds / 60) for minutes, seconds % 60 for remainder, and padStart(2, '0') for two-digit seconds.
  • Electron cross-platform: Electron runs on Windows, macOS, and Linux; it does not run on iOS or iPadOS.
  • Menu removal: mainWindow.setMenu(null) removes the per-window menu on Windows/Linux; Menu.setApplicationMenu(null) removes the app menu on macOS.
  • Path loading: __dirname is the current file's directory; app.getAppPath() is the application root directory (where package.json lives). The latter is safer for packaged apps.

3. Rendering real tracks in the table

// Inside index.html <script>
let allTracks = [];
let tracks = [];

function formatDuration(totalSeconds) {
  if (!totalSeconds || totalSeconds < 0) return '0:00';
  const minutes = Math.floor(totalSeconds / 60);
  const seconds = Math.floor(totalSeconds % 60);
  return `${minutes}:${seconds.toString().padStart(2, '0')}`;
}

function formatArray(arr) {
  if (!arr || arr.length === 0) return '';
  return arr.join(', ');
}

function renderTracks() {
  tbody.innerHTML = '';
  tracks.forEach((track, index) => {
    const tr = document.createElement('tr');
    tr.dataset.id = track.filePath;
    tr.dataset.index = index;
    tr.innerHTML = `
      <td><div class="checkbox"></div></td>
      <td class="track-number">${index + 1}</td>
      <td class="track-title">${track.title}</td>
      <td class="track-artist">${track.artist}</td>
      <td>${track.album}</td>
      <td>${formatArray(track.genre)}</td>
      <td>${formatDuration(track.durationSec)}</td>
    `;
    tbody.appendChild(tr);
  });
}

4. Scan button handler

scanBtn.addEventListener('click', async () => {
  const folderPath = musicPathInput.value.trim();
  try {
    const scanned = await window.electronAPI.scanMusic(folderPath);
    allTracks = scanned;
    tracks = allTracks;
    renderTracks();
  } catch (err) {
    console.error('扫描失败', err);
    alert('扫描失败：' + err.message);
  }
});
